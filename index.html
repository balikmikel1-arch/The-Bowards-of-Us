CREATE OR REPLACE FUNCTION public.af_world_prompt_v53(p_turn integer) RETURNS text LANGUAGE plpgsql IMMUTABLE SET search_path='' AS $$ DECLARE d integer := ((greatest(1,p_turn)-1)/4)+1; BEGIN IF d<=2 THEN RETURN 'Day '||d||'. Some stations experience delays and isolated closures, but most transit still runs. Hospitals are busy, shelves are unevenly stocked, and neighbors disagree about the danger. Avoid sudden citywide ruin.'; ELSIF d<=7 THEN RETURN 'Day '||d||'. Emergency mandates and localized curfews appear. Transit continues on reduced or altered schedules where staff and security allow. A few violent incidents are verified, but most people have never seen an infected person. Police, emergency workers, and utilities are strained, not vanished.'; ELSIF d<=10 THEN RETURN 'Day '||d||'. Neighborhoods diverge: some remain organized while others face roadblocks, supply shortages, and temporary blackouts. Major train lines are suspended in affected areas, while limited service may continue elsewhere. Panic is uneven; no instant blanket abandonment.'; ELSIF d<=14 THEN RETURN 'Day '||d||'. Regional coordination weakens. Shelters and guarded corridors operate, intermittent power and water failures spread, and some neighborhoods are evacuated. Encounters with infected people become more plausible but remain context-dependent. Buildings are recently deserted, not overgrown.'; ELSIF d<=21 THEN RETURN 'Day '||d||'. Emergency government survives in pockets. Most routine public transit has ceased in hard-hit zones; secured evacuation or supply trains may still run. Food, medicine, trust, and local leadership matter. The city shows damage, trash, fires, and neglect, not years of vegetation.'; ELSIF d<=45 THEN RETURN 'Day '||d||'. A fractured city: guarded communities, abandoned blocks, sporadic generators, failed infrastructure, and more frequent dangerous encounters. Survivors remember normal life and respond according to what they have actually experienced. Natural reclamation is still modest.'; ELSE RETURN 'Day '||d||'. Long-term collapse varies by district. Threats grow from scarcity, factions, weather, and infection without constant action-movie hordes.'; END IF; END $$; -- Internal resolver. Only executable by the definer; not exposed to browser RPC. CREATE OR REPLACE FUNCTION public.af_resolve_turn_v52() RETURNS void LANGUAGE plpgsql SECURITY DEFINER SET search_path='' AS $$ DECLARE t integer; humans integer; submitted integer; r record; a jsonb; k text; narrative text; f integer; w integer; med integer; tired integer; hp integer; anxious integer; dlat double precision; dlon double precision; dist double precision; ratio double precision; pos record; dest text; s integer; memory_bias integer; tendency integer; previous text; BEGIN SELECT turn_number INTO t FROM public.af_live_turns WHERE phase='collecting' ORDER BY turn_number DESC LIMIT 1; IF t IS NULL THEN RETURN; END IF; SELECT count(*) INTO humans FROM public.af_session_roster r JOIN public.af_players p ON lower(trim(p.character_name))=lower(trim(r.character_name)) WHERE r.mode='active' AND p.status='active' AND p.health>0 AND p.survey_complete AND lower(trim(r.character_name)) IN ('mikel','avery','daniel','bryce','howard','curtis'); SELECT count(*) INTO submitted FROM public.af_live_actions WHERE turn_number=t; IF submitted < humans THEN RETURN; END IF; -- One deterministic NPC choice per turn, incorporating survey, previous action and survival state. tired:=least(100,tired+15); ELSIF k='rest' THEN tired:=greatest(0,tired-35);anxious:=greatest(0,anxious-12); IF f>0 AND w>0 THEN hp:=least(100,hp+5);END IF; narrative:='You shelter and recover some energy while the outside world changes.'; ELSE 'You cover roughly 25 km toward '||dest||'. The route remains dangerous.' END; ELSE narrative:='You stay alert and assess your surroundings.';tired:=greatest(0,tired-3);END IF; narrative:=narrative||' '||public.af_world_prompt_v53(t); IF f=0 OR w=0 THEN hp:=greatest(0,hp-8);anxious:=least(100,anxious+10);narrative:=narrative||' Shortages are taking a toll.';END IF; IF tired>=90 THEN hp:=greatest(0,hp-4);END IF; UPDATE public.af_survival SET food=f,water=w,medicine=med,fatigue=tired,health=hp,stress=anxious,last_action=k,updated_at=now() WHERE character_name=r.name;
