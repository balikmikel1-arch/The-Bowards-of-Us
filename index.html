-- AFTERFALL V69: active-only queue, automatic NPC resolution, admin reconcile/restart. -- INSTALL THIS FIRST in Supabase SQL Editor. This does not reset the campaign. BEGIN; CREATE OR REPLACE FUNCTION public.af_resolve_turn_v52() RETURNS void LANGUAGE plpgsql SECURITY DEFINER SET search_path TO '' AS $function$ DECLARE t integer; humans integer; submitted integer; loop_row record; a jsonb; k text; narrative text; f integer; w integer; med integer; tired integer; hp integer; anxious integer; dlat double precision; dlon double precision; dist double precision; ratio double precision; pos record; dest text; s integer; memory_bias integer; tendency integer; previous text; BEGIN SELECT turn_number INTO t FROM public.af_live_turns WHERE phase='collecting' ORDER BY turn_number DESC LIMIT 1; IF t IS NULL THEN RETURN; END IF; -- Remove choices submitted by players who are no longer active. DELETE FROM public.af_live_actions a WHERE a.turn_number=t AND NOT EXISTS ( SELECT 1 FROM public.af_players p JOIN public.af_session_roster r ON lower(trim(r.character_name))=lower(trim(p.character_name)) WHERE p.user_id=a.user_id AND r.mode='active' AND p.status='active' AND p.health>0 AND p.survey_complete ); -- Make sure every NPC has survival state before choosing. INSERT INTO public.af_survival(character_name) SELECT r.character_name FROM public.af_session_roster r WHERE r.mode='npc' AND lower(trim(r.character_name)) IN ('mikel','avery','daniel','bryce','howard','curtis') ON CONFLICT DO NOTHING; SELECT count(*) INTO humans FROM public.af_session_roster roster JOIN public.af_players p ON lower(trim(p.character_name))=lower(trim(roster.character_name)) WHERE roster.mode='active' AND p.status='active' AND p.health>0 AND p.survey_complete AND lower(trim(roster.character_name)) IN ('mikel','avery','daniel','bryce','howard','curtis'); SELECT count(*) INTO submitted FROM public.af_live_actions WHERE turn_number=t; IF submitted < humans THEN RETURN; END IF; -- One deterministic NPC choice per turn, incorporating survey, previous action and survival state. FOR loop_row IN SELECT roster.character_name,p.survey,s.food,s.water,s.fatigue,s.stress,s.health,s.last_action FROM public.af_session_roster roster LEFT JOIN public.af_players p ON lower(trim(p.character_name))=lower(trim(roster.character_name)) JOIN public.af_survival s ON s.character_name=roster.character_name WHERE roster.mode='npc' AND lower(trim(roster.character_name)) IN ('mikel','avery','daniel','bryce','howard','curtis') LOOP IF NOT EXISTS(SELECT 1 FROM public.af_npc_memory WHERE character_name=loop_row.character_name AND turn_number=t) THEN -- Stable per-character survey-derived bias. This is a heuristic, not AI comprehension of answers. tendency:=mod(abs(hashtext(loop_row.character_name||coalesce(loop_row.survey::text,''))),100); memory_bias:=CASE loop_row.last_action WHEN 'scavenge' THEN 12 WHEN 'rest' THEN -12 ELSE 0 END; IF loop_row.water<=1 OR loop_row.food<=1 THEN k:='scavenge';narrative:='Supplies are running low.'; ELSIF loop_row.fatigue>=60 OR loop_row.health<=50 THEN k:='rest';narrative:='Exhaustion or injury demands recovery.'; ELSIF loop_row.stress>=75 THEN k:='hold';narrative:='High stress encourages caution.'; ELSIF tendency+memory_bias>=56 THEN k:='scavenge';narrative:='Survey-shaped risk tolerance and past behavior favor searching.'; ELSIF tendency<30 THEN k:='rest';narrative:='Survey-shaped caution favors conserving energy.'; ELSE k:='hold';narrative:='Survey-shaped temperament favors watching and listening.';END IF; INSERT INTO public.af_npc_memory(character_name,turn_number,action,reason) VALUES(loop_row.character_name,t,jsonb_build_object('kind',k),narrative); END IF; END LOOP; -- Resolve active human actions and server-created NPC actions in a single transaction. FOR loop_row IN SELECT p.character_name AS name,a.user_id,a.action FROM public.af_live_actions a JOIN public.af_players p ON p.user_id=a.user_id WHERE a.turn_number=t UNION ALL SELECT m.character_name AS name,NULL::uuid,m.action FROM public.af_npc_memory m WHERE m.turn_number=t LOOP a:=loop_row.action;k:=a->>'kind'; INSERT INTO public.af_survival(character_name) VALUES(loop_row.name) ON CONFLICT DO NOTHING; SELECT food,water,medicine,fatigue,health,stress INTO f,w,med,tired,hp,anxious FROM public.af_survival WHERE character_name=loop_row.name FOR UPDATE; -- Baseline resource usage per turn, bounded at zero. f:=greatest(0,f-1);w:=greatest(0,w-1); IF k='scavenge' THEN -- Stable pseudo-randomized loot, generated on the server once per action. s:=mod(abs(hashtext(loop_row.name||':'||t::text||':'||coalesce(a::text,''))),100); IF s<65 THEN f:=f+2;w:=w+2;narrative:='You locate food and drinking water in a careful search.'; ELSIF s<82 THEN med:=med+1;narrative:='You discover a small medical kit.'; ELSE narrative:='The search turns up nothing useful, and the noise puts you on edge.';anxious:=least(100,anxious+12);END IF; tired:=least(100,tired+15); ELSIF k='rest' THEN tired:=greatest(0,tired-35);anxious:=greatest(0,anxious-12); IF f>0 AND w>0 THEN hp:=least(100,hp+5);END IF; narrative:='You shelter and recover some energy while the outside world changes.'; ELSIF k LIKE 'travel:%' THEN -- Travel uses bounded progress; it is NOT turn-by-turn road routing. dest:=a->'target'->>'label';dlat:=(a->'target'->>'lat')::double precision;dlon:=(a->'target'->>'lon')::double precision; SELECT lat,lon,label INTO pos FROM public.af_live_positions WHERE user_id=loop_row.user_id; IF pos.lat IS NULL THEN RAISE EXCEPTION 'No saved travel position for %',loop_row.name;END IF; dist:=111.2*sqrt(power(dlat-pos.lat,2)+power((dlon-pos.lon)*cos(radians(pos.lat)),2)); ratio:=CASE WHEN dist<=25 THEN 1 ELSE 25/dist END; UPDATE public.af_live_positions SET lat=pos.lat+(dlat-pos.lat)*ratio,lon=pos.lon+(dlon-pos.lon)*ratio, label=CASE WHEN ratio=1 THEN dest ELSE 'On the road toward '||dest END, travel_to=CASE WHEN ratio=1 THEN NULL ELSE dest END,updated_at=now() WHERE user_id=loop_row.user_id; tired:=least(100,tired+24);anxious:=least(100,anxious+8); narrative:=CASE WHEN ratio=1 THEN 'You reach '||dest||'.' ELSE 'You cover roughly 25 km toward '||dest||'. The route remains dangerous.' END; ELSE narrative:='You stay alert and assess your surroundings.';tired:=greatest(0,tired-3);END IF; narrative:=narrative||' '||public.af_world_prompt_v53(t); IF f=0 OR w=0 THEN hp:=greatest(0,hp-8);anxious:=least(100,anxious+10);narrative:=narrative||' Shortages are taking a toll.';END IF; IF tired>=90 THEN hp:=greatest(0,hp-4);END IF; UPDATE public.af_survival SET food=f,water=w,medicine=med,fatigue=tired,health=hp,stress=anxious,last_action=k,updated_at=now() WHERE character_name=loop_row.name; IF loop_row.user_id IS NOT NULL THEN INSERT INTO public.af_live_journal(user_id,turn_number,body) VALUES(loop_row.user_id,t,narrative);END IF; END LOOP; UPDATE public.af_live_turns SET phase='resolved',resolved_at=now() WHERE turn_number=t; INSERT INTO public.af_live_turns(turn_number,title,prompt) VALUES(t+1,'The World Moves · Turn '||(t+1), public.af_world_prompt_v53(t+1)); END $function$; CREATE OR REPLACE FUNCTION public.af_live_snapshot() RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path TO '' AS $function$ DECLARE t public.af_live_turns%rowtype;p public.af_live_positions%rowtype; n integer;submitted integer;mine boolean; v_player_mode text; charname text; stats jsonb; BEGIN IF auth.uid() IS NULL THEN RAISE EXCEPTION 'Sign in first.';END IF; SELECT character_name INTO charname FROM public.af_players WHERE user_id=auth.uid() AND survey_complete; IF charname IS NULL THEN RAISE EXCEPTION 'Complete your survivor survey to enter the live world.';END IF; SELECT r.mode INTO v_player_mode FROM public.af_session_roster r WHERE lower(trim(r.character_name))=lower(trim(charname)); IF v_player_mode IS NULL THEN RAISE EXCEPTION 'Character is not in the six-player roster.';END IF; SELECT * INTO t FROM public.af_live_turns ORDER BY turn_number DESC LIMIT 1; SELECT * INTO p FROM public.af_live_positions WHERE user_id=auth.uid(); SELECT count(*) INTO n FROM public.af_session_roster r JOIN public.af_players p ON lower(trim(p.character_name))=lower(trim(r.character_name)) WHERE r.mode='active' AND p.status='active' AND p.health>0 AND p.survey_complete AND lower(trim(r.character_name)) IN ('mikel','avery','daniel','bryce','howard','curtis'); IF t.turn_number IS NOT NULL THEN SELECT count(*) INTO submitted FROM public.af_live_actions a JOIN public.af_players p ON p.user_id=a.user_id JOIN public.af_session_roster r ON lower(trim(r.character_name))=lower(trim(p.character_name)) WHERE a.turn_number=t.turn_number AND r.mode='active' AND p.status='active' AND p.health>0 AND p.survey_complete; SELECT EXISTS(SELECT 1 FROM public.af_live_actions WHERE turn_number=t.turn_number AND user_id=auth.uid()) INTO mine; ELSE submitted:=0;mine:=false;END IF; SELECT to_jsonb(s)-'character_name'-'updated_at' INTO stats FROM public.af_survival s WHERE lower(trim(s.character_name))=lower(trim(charname)); RETURN jsonb_build_object('turn',CASE WHEN t.turn_number IS NULL THEN NULL ELSE jsonb_build_object('turn_number',t.turn_number,'phase',t.phase,'title',t.title,'prompt',t.prompt) END, 'position',CASE WHEN p.user_id IS NULL THEN NULL ELSE jsonb_build_object('lat',p.lat,'lon',p.lon,'label',p.label,'travel_to',p.travel_to) END, 'participants',n,'submitted',submitted,'submitted_by_me',mine,'my_mode',v_player_mode,'survival',coalesce(stats,'{}'::jsonb), 'perspective',CASE WHEN p.user_id IS NULL THEN 'The world is waiting to begin.' ELSE 'You are in '||p.label||'. You can only act on what your survivor can see and hear.' END, 'events',coalesce((SELECT jsonb_agg(to_jsonb(j)) FROM (SELECT turn_number,body FROM public.af_live_journal WHERE user_id=auth.uid() ORDER BY id DESC LIMIT 12) j),'[]'::jsonb)); END $function$; CREATE OR REPLACE FUNCTION public.af_admin_reconcile_v69() RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path TO '' AS $v69$ DECLARE before_turn integer; after_turn integer; needed integer; ready integer; BEGIN IF auth.uid() IS NULL OR NOT EXISTS ( SELECT 1 FROM public.af_players p WHERE p.user_id=auth.uid() AND lower(trim(p.character_name))='mikel' AND p.survey_complete ) THEN RAISE EXCEPTION 'Only Mikel can reconcile the campaign.'; END IF; PERFORM pg_advisory_xact_lock(491491); SELECT max(turn_number) INTO before_turn FROM public.af_live_turns; PERFORM public.af_resolve_turn_v52(); SELECT max(turn_number) INTO after_turn FROM public.af_live_turns; SELECT count(*) INTO needed FROM public.af_session_roster r JOIN public.af_players p ON lower(trim(p.character_name))=lower(trim(r.character_name)) WHERE r.mode='active' AND p.status='active' AND p.health>0 AND p.survey_complete AND lower(trim(r.character_name)) IN ('mikel','avery','daniel','bryce','howard','curtis'); SELECT count(*) INTO ready FROM public.af_live_actions a JOIN public.af_players p ON p.user_id=a.user_id JOIN public.af_session_roster r ON lower(trim(r.character_name))=lower(trim(p.character_name)) WHERE a.turn_number=after_turn AND r.mode='active' AND p.status='active' AND p.health>0 AND p.survey_complete; RETURN jsonb_build_object('advanced',coalesce(after_turn>before_turn,false),'ready',ready,'needed',needed); END $v69$; CREATE OR REPLACE FUNCTION public.af_admin_restart_v69() RETURNS void LANGUAGE plpgsql SECURITY DEFINER SET search_path TO '' AS $v69$ BEGIN IF auth.uid() IS NULL OR NOT EXISTS ( SELECT 1 FROM public.af_players p WHERE p.user_id=auth.uid() AND lower(trim(p.character_name))='mikel' AND p.survey_complete ) THEN RAISE EXCEPTION 'Only Mikel can restart the campaign.'; END IF; PERFORM pg_advisory_xact_lock(491491); IF to_regclass('public.af_ai_decisions') IS NOT NULL THEN EXECUTE 'DELETE FROM public.af_ai_decisions'; END IF; IF to_regclass('public.af_ai_scenes') IS NOT NULL THEN EXECUTE 'DELETE FROM public.af_ai_scenes'; END IF; PERFORM public.af_admin_restart_campaign(); END $v69$; GRANT EXECUTE ON FUNCTION public.af_admin_reconcile_v69() TO authenticated; GRANT EXECUTE ON FUNCTION public.af_admin_restart_v69() TO authenticated; COMMIT;
